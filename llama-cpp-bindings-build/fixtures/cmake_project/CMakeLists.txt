cmake_minimum_required(VERSION 3.14)
project(llama_cpp_bindings_build_fixture C)

if(NOT DEFINED FIXTURE_LIBRARY_NAME)
    message(FATAL_ERROR "FIXTURE_LIBRARY_NAME must be defined")
endif()

add_library(${FIXTURE_LIBRARY_NAME} STATIC fixture.c)

install(TARGETS ${FIXTURE_LIBRARY_NAME} DESTINATION lib)

set(GGML_OPENMP_ENABLED "ON" CACHE INTERNAL "")
set(OpenMP_C_LIB_NAMES "${FIXTURE_LIBRARY_NAME}" CACHE STRING "")
set(OpenMP_CXX_LIB_NAMES "${FIXTURE_LIBRARY_NAME}" CACHE STRING "")
set(OpenMP_${FIXTURE_LIBRARY_NAME}_LIBRARY
    "${CMAKE_INSTALL_PREFIX}/lib/${CMAKE_STATIC_LIBRARY_PREFIX}${FIXTURE_LIBRARY_NAME}${CMAKE_STATIC_LIBRARY_SUFFIX}"
    CACHE FILEPATH "")
